    18→    18→function parse<T>(raw: string | null | undefined, fallback: T): T {
    19→    19→  if (!raw) return fallback
    20→    20→  try {
    21→    21→    return JSON.parse(raw) as T
    22→    22→  } catch {
    23→    23→    return fallback
    24→    24→  }

    27→    27→function mermaidErd(entities: Array<{ name: string; relationships: string | null }>): string {
    28→    28→  const rels = entities.flatMap((e) =>
    29→    29→    parse<Array<{ target: string; type: string }>>(e.relationships, []).map((r) => ({ from: e.name, ...r }))
    30→    30→  )
    31→    31→  const sanitize = (n: string) => n.replace(/[^a-zA-Z0-9]/g, '_')
    32→    32→  const lines = ['erDiagram']
    33→    33→  for (const e of entities) {
    34→    34→    lines.push(`  ${sanitize(e.name)} {`)
    35→    35→    lines.push('    string id PK')
    36→    36→    lines.push('  }')
    37→    37→  }
    38→    38→  for (const r of rels) {
    39→    39→    const cardinality =
    40→    40→      r.type.includes('many-to-many')
    41→    41→        ? '}o..o{'
    42→    42→        : r.type.includes('one-to-many')
    43→    43→          ? '||--o{'
    44→    44→          : r.type.includes('one-to-one')
    45→    45→            ? '||--||'
    46→    46→            : '||--o{'
    47→    47→    lines.push(`  ${sanitize(r.from)} ${cardinality} ${sanitize(r.target)} : "${r.type}"`)
    48→    48→  }
    49→    49→  return lines.join('\n')
    50→    50→}
