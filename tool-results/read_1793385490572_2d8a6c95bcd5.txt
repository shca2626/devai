    22→    22→// Express autopilot — the client's explicit DELEGATION of the
    23→    23→// remaining workflow to the virtual engineer. One unit of work
    24→    24→// per call; the UI drives the loop and shows live progress.
    25→    25→//
    26→    26→// Philosophy preserved:
    27→    27→// - Delegation is itself an explicit, logged act (never silence).
    28→    28→// - Every auto-decision is written to the decision history with
    29→    29→//   an Express marker and can be changed afterwards — the
    30→    30→//   package can be reopened, artifacts edited, nothing restarts.

   113→   113→async function autoAnswerQuestions(ideaId: string): Promise<number> {
   114→   114→  const pending = await db.question.findMany({
   115→   115→    where: { ideaId, status: 'pending' },
   116→   116→    orderBy: [{ order: 'asc' }, { createdAt: 'asc' }],
   117→   117→  })
   118→   118→  if (pending.length === 0) return 0
   119→   119→
   120→   120→  const idea = await db.idea.findUnique({ where: { id: ideaId } })
   121→   121→  const interpretation = parseInterpretation(idea?.interpretation ?? null)
   122→   122→  const ctx: ProjectContext | null = await buildProjectContext(ideaId)
   123→   123→
   124→   124→  const lines = pending.map((q, i) => {
   125→   125→    const options = safeArr(q.options).length > 0 ? safeArr(q.options) : []
   126→   126→    const optStr = options
   127→   127→      .map((o) => {
   128→   128→        const m = o as { label?: string; consequence?: string }
   129→   129→        return `  - "${m.label ?? ''}"${m.consequence ? ` (consequence: ${m.consequence})` : ''}`
   130→   130→      })
   131→   131→      .join('\n')
   132→   132→    return `Q${i + 1} [${q.area}] ${q.questionEn}\nwhy it matters: ${q.whyItMatters ?? '-'}\noptions:\n${optStr || '  (no predefined options — answer freely, max 12 words)'}`
   133→   133→  })
   134→   134→
   135→   135→  const answered = await answeredItems(ideaId)
   136→   136→  const answeredBlock =
   137→   137→    answered.length > 0
   138→   138→      ? `\nAlready answered by the client earlier (respect these as facts):\n${answered.map((a) => `- ${a.questionEn} → ${a.answer}`).join('\n')}`
   139→   139→      : ''
   140→   140→
   141→   141→  const raw = await chatJson<{ answers?: Array<{ n?: number; choice?: string; why?: string }> }>(
   142→   142→    AUTO_ANSWER_PROMPT,
   143→   143→    `${ctx ? '' : ''}${interpretation ? `Product interpretation:\n${interpretation.englishInterpretation}\nProblem: ${interpretation.problemStatement}\n\n` : ''}Pending questions:\n${lines.join('\n\n')}${answeredBlock}`
   144→   144→  )
   145→   145→
   146→   146→  const answers = Array.isArray(raw?.answers) ? raw.answers : []
   147→   147→  let count = 0
   148→   148→  for (let i = 0; i < pending.length; i++) {
   149→   149→    const match = answers.find((a) => Number(a.n) === i + 1)
   150→   150→    const fallbackOptions = safeArr(pending[i].options)
   151→   151→    const fallbackLabel =
   152→   152→      (fallbackOptions[0] as { label?: string } | undefined)?.label ?? 'Decided by the engineer (express)'
   153→   153→    const answer = typeof match?.choice === 'string' && match.choice.trim() ? match.choice.trim() : fallbackLabel
   154→   154→    await db.question.update({
   155→   155→      where: { id: pending[i].id },
   156→   156→      data: { answer, answerKind: 'choice', status: 'answered' },
   157→   157→    })
   158→   158→    // Every auto-answer is recorded as an OPEN assumption — never a silent fact.
   159→   159→    await db.assumption.create({
   160→   160→      data: {
   161→   161→        ideaId,
   162→   162→        text: `Express auto-answer — Q: ${pending[i].questionEn} → A: ${answer}`,
   163→   163→        kind: 'suggested',
   164→   164→        status: 'open',
   165→   165→        source: 'express',
   166→   166→      },
   167→   167→    })
   168→   168→    count++
   169→   169→  }
   170→   170→  return count
   171→   171→}

   189→   189→async function autoFixCriticalIssues(
   190→   190→  ideaId: string,
   191→   191→  issues: Array<{ id: string; description: string; affected: string | null }>
   192→   192→): Promise<number> {
   193→   193→  const [entities, apis] = await Promise.all([
   194→   194→    db.dataEntity.findMany({ where: { ideaId } }),
   195→   195→    db.apiEndpoint.findMany({ where: { ideaId } }),
   196→   196→  ])
   197→   197→
   198→   198→  const issueBlock = issues
   199→   199→    .map((i, idx) => `Issue #${idx}: ${i.description}${i.affected ? ` (affected: ${i.affected})` : ''}`)
   200→   200→    .join('\n')
   201→   201→  const entityBlock = entities
   202→   202→    .map((e) => `- ${e.code} "${e.name}": fields = ${e.fields}`)
   203→   203→    .join('\n')
   204→   204→  const apiBlock = apis
   205→   205→    .map((a) => `- ${a.code} ${a.method} ${a.path}: request = ${a.request ?? '[]'}`)
   206→   206→    .join('\n')
   207→   207→
   208→   208→  const raw = await chatJson<{ fixes?: Array<Record<string, unknown>> }>(
   209→   209→    AUTOFIX_PROMPT,
   210→   210→    `Critical issues:\n${issueBlock}\n\nCurrent data entities:\n${entityBlock}\n\nCurrent API endpoints:\n${apiBlock}`
   211→   211→  )
   212→   212→
   213→   213→  const fixes = Array.isArray(raw?.fixes) ? raw.fixes : []
   214→   214→  const parseJsonArr = (s: string | null): Array<Record<string, unknown>> => {
   215→   215→    try {
   216→   216→      const v = s ? JSON.parse(s) : []
   217→   217→      return Array.isArray(v) ? v : []
   218→   218→    } catch {
   219→   219→      return []
   220→   220→    }
   221→   221→  }
   222→   222→
   223→   223→  let applied = 0
   224→   224→  const appliedNotes: string[] = []
   225→   225→  const norm = (s: string) => s.trim().toLowerCase()
   226→   226→
   227→   227→  for (const f of fixes) {
   228→   228→    const op = typeof f.op === 'string' ? f.op : ''
   229→   229→    const note = typeof f.note === 'string' ? f.note : ''
   230→   230→    const issueIndex = typeof f.issueIndex === 'number' ? f.issueIndex : -1
   231→   231→    const issue = issueIndex >= 0 && issueIndex < issues.length ? issues[issueIndex] : null
   232→   232→
   233→   233→    try {
   234→   234→      if (op === 'add_entity_field' || op === 'remove_entity_field') {
   235→   235→        const ref = String(f.entity ?? '')
   236→   236→        const target = entities.find((e) => norm(e.name) === norm(ref) || norm(e.code) === norm(ref))
   237→   237→        if (!target) continue
   238→   238→        const fields = parseJsonArr(target.fields)
   239→   239→        let next: Array<Record<string, unknown>>
   240→   240→        if (op === 'add_entity_field') {
   241→   241→          const fm = (f.field ?? {}) as Record<string, unknown>
   242→   242→          const name = typeof fm.name === 'string' ? fm.name.trim() : ''
   243→   243→          if (!name) continue
   244→   244→          if (fields.some((x) => x.name === name)) {
   245→   245→            // The data is already consistent — the issue row is stale.
   246→   246→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   247→   247→            applied++
   248→   248→            appliedNotes.push(`${target.code}: field "${name}" already present — stale issue resolved`)
   249→   249→            continue
   250→   250→          }
   251→   251→          next = [...fields, {
   252→   252→            name,
   253→   253→            type: typeof fm.type === 'string' && fm.type.trim() ? fm.type.trim() : 'string',
   254→   254→            required: fm.required === true,
   255→   255→            unique: fm.unique === true,
   256→   256→            description: typeof fm.description === 'string' ? fm.description : '',
   257→   257→          }]
   258→   258→        } else {
   259→   259→          const fname = String(f.field ?? '').trim()
   260→   260→          next = fields.filter((x) => x.name !== fname)
   261→   261→          if (next.length === fields.length) {
   262→   262→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   263→   263→            applied++
   264→   264→            appliedNotes.push(`${target.code}: field "${fname}" already absent — stale issue resolved`)
   265→   265→            continue
   266→   266→          }
   267→   267→        }
   268→   268→        await db.dataEntity.update({
   269→   269→          where: { id: target.id },
   270→   270→          data: {
   271→   271→            fields: JSON.stringify(next),
   272→   272→            status: target.status === 'confirmed' ? 'confirmed' : target.status,
   273→   273→          },
   274→   274→        })
   275→   275→        applied++
   276→   276→        appliedNotes.push(`${target.code}: ${op === 'add_entity_field' ? 'added' : 'removed'} field — ${note}`)
   277→   277→        if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   278→   278→      } else if (op === 'add_api_request_field' || op === 'remove_api_request_field') {
   279→   279→        const ref = String(f.api ?? '')
   280→   280→        const target = apis.find(
   281→   281→          (a) => norm(a.code) === norm(ref) || norm(a.path) === norm(ref) || norm(`${a.method} ${a.path}`) === norm(ref)
   282→   282→        )
   283→   283→        if (!target) continue
   284→   284→        const request = parseJsonArr(target.request)
   285→   285→        let next: Array<Record<string, unknown>>
   286→   286→        if (op === 'add_api_request_field') {
   287→   287→          const fm = (f.field ?? {}) as Record<string, unknown>
   288→   288→          const field = typeof fm.field === 'string' ? fm.field.trim() : ''
   289→   289→          if (!field) continue
   290→   290→          if (request.some((x) => x.field === field)) {
   291→   291→            // Already consistent — the issue row is stale.
   292→   292→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   293→   293→            applied++
   294→   294→            appliedNotes.push(`${target.code}: request field "${field}" already present — stale issue resolved`)
   295→   295→            continue
   296→   296→          }
   297→   297→          next = [...request, {
   298→   298→            field,
   299→   299→            type: typeof fm.type === 'string' && fm.type.trim() ? fm.type.trim() : 'string',
   300→   300→            required: fm.required === true,
   301→   301→            notes: typeof fm.notes === 'string' ? fm.notes : '',
   302→   302→          }]
   303→   303→        } else {
   304→   304→          const fname = String(f.field ?? '').trim()
   305→   305→          next = request.filter((x) => x.field !== fname)
   306→   306→          if (next.length === request.length) {
   307→   307→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   308→   308→            applied++
   309→   309→            appliedNotes.push(`${target.code}: request field "${fname}" already absent — stale issue resolved`)
   310→   310→            continue
   311→   311→          }
   312→   312→        }
   313→   313→        await db.apiEndpoint.update({
   314→   314→          where: { id: target.id },
   315→   315→          data: {
   316→   316→            request: JSON.stringify(next),
   317→   317→            status: target.status === 'confirmed' ? 'confirmed' : target.status,
   318→   318→          },
   319→   319→        })
   320→   320→        applied++
   321→   321→        appliedNotes.push(`${target.code}: ${op === 'add_api_request_field' ? 'added' : 'removed'} request field — ${note}`)
   322→   322→        if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   323→   323→      }
   324→   324→    } catch {
   325→   325→      // a single malformed fix must not abort the others
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  if (applied > 0) {
   330→   330→    await logDecision(
   331→   331→      ideaId,
   332→   332→      'Express: critical issues auto-fixed',
   333→   333→      appliedNotes.slice(0, 5).join(' | '),
   334→   334→      'readiness'
   335→   335→    )
   336→   336→  }
   337→   337→  return applied
   338→   338→}

   351→   351→async function resolveUnresolvedMusts(ideaId: string): Promise<number> {
   352→   352→  const rows = await db.requirement.findMany({ where: { ideaId, status: 'unresolved', priority: 'must' } })
   353→   353→  if (rows.length === 0) return 0
   354→   354→
   355→   355→  const confirmed = await db.requirement.findMany({
   356→   356→    where: { ideaId, status: 'confirmed' },
   357→   357→    select: { code: true, title: true },
   358→   358→  })
   359→   359→
   360→   360→  let decisions: Array<{ code?: string; decision?: string; why?: string }> = []
   361→   361→  try {
   362→   362→    const raw = await chatJson<{ decisions?: Array<{ code?: string; decision?: string; why?: string }> }>(
   363→   363→      RESOLVE_MUST_PROMPT,
   364→   364→      `Unresolved must-have requirements:
   365→   365→${rows.map((r) => `- ${r.code}: ${r.title}\n  ${r.description}\n  evidence: ${r.evidence ?? '-'}`).join('\n')}
   366→   366→
   367→   367→Confirmed requirements (facts):
   368→   368→${confirmed.map((c) => `- ${c.code}: ${c.title}`).join('\n')}`
   369→   369→    )
   370→   370→    decisions = Array.isArray(raw?.decisions) ? raw.decisions : []
   371→   371→  } catch {
   372→   372→    decisions = []
   373→   373→  }
   374→   374→
   375→   375→  let resolved = 0
   376→   376→  for (const r of rows) {
   377→   377→    const d = decisions.find((x) => x.code === r.code)
   378→   378→    const decision = d?.decision === 'rejected' ? 'rejected' : 'confirmed'
   379→   379→    const why = typeof d?.why === 'string' ? d.why : 'delegated fallback — confirmed to keep the must-have scope explicit'
   380→   380→    await db.requirement.update({ where: { id: r.id }, data: { status: decision } })
   381→   381→    await logDecision(ideaId, `Express: must-have ${r.code} → ${decision}`, why, 'readiness')
   382→   382→    resolved++
   383→   383→  }
   384→   384→  return resolved
   385→   385→}

   389→   389→export async function runExpressStep(ideaId: string): Promise<ExpressStepResult> {
   390→   390→  const idea = await db.idea.findUnique({ where: { id: ideaId } })
   391→   391→  if (!idea) return blocked('Idea not found.')
   392→   392→  const id = ideaId
   393→   393→
   394→   394→  // ===== Stage: understanding =====
   395→   395→  if (idea.stage === 'understanding') {
   396→   396→    const interpretation = parseInterpretation(idea.interpretation)
   397→   397→    if (!interpretation) {
   398→   398→      return blocked('There is no interpretation yet — describe the idea and run the understanding step first.')
   399→   399→    }
   400→   400→    const drafts = await generateQuestions({ interpretation, answered: [], uncoveredAreas: [...AREAS], weakAreas: [] })
   401→   401→    let order = 0
   402→   402→    for (const draft of drafts) {
   403→   403→      await db.question.create({
   404→   404→        data: {
   405→   405→          ideaId: id,
   406→   406→          area: draft.area,
   407→   407→          questionEn: draft.questionEn,
   408→   408→          questionAr: draft.questionAr,
   409→   409→          whyItMatters: draft.whyItMatters,
   410→   410→          options: JSON.stringify(draft.options),
   411→   411→          status: 'pending',
   412→   412→          order: order++,
   413→   413→        },
   414→   414→      })

   816→   816→      const drafts = await generateRisks({ ctx, requirements: await loadRequirements(id) })
   817→   817→      let count = 0
   818→   818→      for (const d of drafts) {
   819→   819→        await db.riskItem.create({
   820→   820→          data: {
   821→   821→            ideaId: id,
   822→   822→            code: `RSK-${String(++count).padStart(2, '0')}`,
   823→   823→            category: d.category,
   824→   824→            description: d.description,
   825→   825→            likelihood: d.likelihood,
   826→   826→            impact: d.impact,
   827→   827→            mitigation: d.mitigation,
   828→   828→            basis: d.basis,
   829→   829→            status: 'proposed',
   830→   830→            order: count - 1,
   831→   831→          },
   832→   832→        })
   833→   833→      }
   834→   834→      await logDecision(id, 'Express: risk register generated', `${count} risks`, 'planning')
   835→   835→      return running(`Risk register generated (${count} risks)`, 'Deciding the risk mitigations')
   836→   836→    }
   837→   837→
   838→   838→    const riskOpen = risks.filter((r) => r.status === 'proposed')
   839→   839→    if (riskOpen.length > 0) {
   840→   840→      await db.riskItem.updateMany({ where: { ideaId: id, status: 'proposed' }, data: { status: 'mitigated' } })
   841→   841→      await logDecision(id, 'Express: risks mitigated', `${riskOpen.length} mitigated by delegation`, 'planning')
   842→   842→      return running(`Mitigated ${riskOpen.length} risks`, 'Approving the plan & checking readiness')
   843→   843→    }
   844→   844→
   845→   845→    await db.idea.update({ where: { id }, data: { stage: 'readiness' } })
   846→   846→    await logDecision(id, 'Express: implementation plan approved', 'Tasks and risks reviewed', 'planning')
   847→   847→    return running('Implementation plan approved', 'Running the consistency validation')
   848→   848→  }
   849→   849→
   850→   850→  // ===== Stage: readiness =====
   851→   851→  if (idea.stage === 'readiness') {
   852→   852→    // Run consistency once (detected via the decision history)
   853→   853→    const decisions = await db.decision.findMany({
   854→   854→      where: { ideaId: id, action: { in: ['Consistency validation executed', 'Express: consistency validation executed'] } },
   855→   855→      select: { id: true },
   856→   856→    })
   857→   857→    if (decisions.length === 0) {
   858→   858→      const ctx = await buildProjectContext(id)
   859→   859→      if (!ctx) return blocked('Idea not found.')
   860→   860→      const detected = await runConsistencyCheck({ ideaId: id, ctx })
   861→   861→      await db.consistencyIssue.deleteMany({ where: { ideaId: id, status: 'open', detectedBy: { in: ['engine', 'ai'] } } })
   862→   862→      for (const issue of detected) {
   863→   863→        await db.consistencyIssue.create({
   864→   864→          data: {
   865→   865→            ideaId: id,
   866→   866→            kind: issue.kind,
   867→   867→            severity: issue.severity,
   868→   868→            description: issue.description,
   869→   869→            affected: JSON.stringify(issue.affected),
   870→   870→            suggestion: issue.suggestion,
   871→   871→            detectedBy: issue.detectedBy,
   872→   872→            status: 'open',
   873→   873→          },
   874→   874→        })
   875→   875→      }
   876→   876→      const critical = detected.filter((i) => i.severity === 'critical').length
   877→   877→      const major = detected.filter((i) => i.severity === 'major').length
   878→   878→      await logDecision(
   879→   879→        id,
   880→   880→        'Express: consistency validation executed',
   881→   881→        detected.length === 0 ? 'No issues found' : `${detected.length} issue(s) detected (${critical} critical, ${major} major)`,
   882→   882→        'readiness'
   883→   883→      )
   884→   884→      return running(
   885→   885→        `Consistency validation done — ${detected.length} issue(s) found (${critical} critical)`,
   886→   886→        'Handling consistency issues'
   887→   887→      )
   888→   888→    }
   889→   889→
   890→   890→    // Critical issues: the engineer FIRST tries to fix them himself
   891→   891→    // (delegated authority); only what is genuinely unfixable is
   892→   892→    // escalated to the client.
   893→   893→    const openCritical = await db.consistencyIssue.findMany({ where: { ideaId: id, status: 'open', severity: 'critical' } })
   894→   894→    if (openCritical.length > 0) {
   895→   895→      const fixed = await autoFixCriticalIssues(
   896→   896→        id,
   897→   897→        openCritical.map((i) => ({ id: i.id, description: i.description, affected: i.affected }))
   898→   898→      )
   899→   899→      if (fixed > 0) {
   900→   900→        return running(
   901→   901→          `Auto-fixed ${fixed} critical consistency issue(s) — the artifacts were updated and the fix is logged`,
   902→   902→          'Re-checking critical issues'
   903→   903→        )
   904→   904→      }
   905→   905→      return blocked(
   906→   906→        'Critical consistency issues need a human decision before the package can be locked. Resolve or dismiss them in the readiness stage, then re-run express.',
   907→   907→        openCritical.map((i) => i.description)
   908→   908→      )
   909→   909→    }
   910→   910→
   911→   911→    // Minor issues are dismissed by delegation; major ones stay visible as warnings.
   912→   912→    const openMinor = await db.consistencyIssue.findMany({ where: { ideaId: id, status: 'open', severity: 'minor' } })
   913→   913→    if (openMinor.length > 0) {
   914→   914→      await db.consistencyIssue.updateMany({ where: { ideaId: id, status: 'open', severity: 'minor' }, data: { status: 'dismissed' } })
   915→   915→      await logDecision(id, 'Express: minor issues dismissed', `${openMinor.length} dismissed by delegation (major issues stay visible)`, 'readiness')
   916→   916→      return running(`Dismissed ${openMinor.length} minor consistency issues`, 'Checking the readiness gate')
   917→   917→    }
   918→   918→
   919→   919→    // Unresolved must-have requirements are decided by the engineer
   920→   920→    // under delegation — each decision is logged with its reasoning.
   921→   921→    const unresolvedMust = await db.requirement.count({
   922→   922→      where: { ideaId: id, status: 'unresolved', priority: 'must' },
   923→   923→    })
   924→   924→    if (unresolvedMust > 0) {
   925→   925→      const n = await resolveUnresolvedMusts(id)
   926→   926→      return running(
   927→   927→        `Decided ${n} open must-have requirement(s) — each decision logged with its reasoning`,
   928→   928→        'Checking the readiness gate'
   929→   929→      )
   930→   930→    }
   931→   931→
   932→   932→    // Coverage gap: confirmed FRs without a confirmed task (typically
   933→   933→    // requirements confirmed after the plan was drafted). The engineer
   934→   934→    // drafts targeted tasks for exactly the uncovered requirements.
   935→   935→    const [confirmedTasks, confirmedFRs, confirmedEntities, confirmedApis, confirmedScreens] = await Promise.all([
   936→   936→      db.implTask.findMany({ where: { ideaId: id, status: 'confirmed' } }),
   937→   937→      db.requirement.findMany({ where: { ideaId: id, status: 'confirmed', type: 'functional' } }),
   938→   938→      db.dataEntity.findMany({ where: { ideaId: id, status: 'confirmed' } }),
   939→   939→      db.apiEndpoint.findMany({ where: { ideaId: id, status: 'confirmed' } }),
   940→   940→      db.screenSpec.findMany({ where: { ideaId: id, status: 'confirmed' } }),
   941→   941→    ])
   942→   942→    const taskLinks = new Set(confirmedTasks.flatMap((t) => safeArr(t.requirementCodes)))
   943→   943→    const uncovered = confirmedFRs.filter((r) => !taskLinks.has(r.code))
   944→   944→    if (uncovered.length > 0) {
   945→   945→      const ctx = await buildProjectContext(id)
   946→   946→      if (!ctx) return blocked('Idea not found.')
   947→   947→      const uncoveredSet = new Set(uncovered.map((u) => u.code))
   948→   948→      const drafts = await generateTasks({
   949→   949→        ctx,
   950→   950→        requirements: await loadRequirements(id),
   951→   951→        entityNames: confirmedEntities.map((e) => e.name),
   952→   952→        apiPaths: confirmedApis.map((a) => `${a.method} ${a.path}`),
   953→   953→        screenNames: confirmedScreens.map((s) => s.name),
   954→   954→      })
   955→   955→      const targeted = drafts.filter((d) => d.requirementCodes.some((c) => uncoveredSet.has(c))).slice(0, 8)
   956→   956→      if (targeted.length === 0) {
   957→   957→        return blocked(
   958→   958→          'These confirmed requirements have no implementation task and the engineer could not draft one automatically:',
   959→   959→          uncovered.map((u) => `${u.code}: ${u.title}`)
   960→   960→        )
   961→   961→      }
   962→   962→      const existingCount = await db.implTask.count({ where: { ideaId: id } })
   963→   963→      let n = 0
   964→   964→      const idToCode = new Map<string, string>()
   965→   965→      for (const d of targeted) idToCode.set(d.id, `T-${String(existingCount + idToCode.size + 1).padStart(2, '0')}`)
   966→   966→      for (const d of targeted) {
   967→   967→        await db.implTask.create({
   968→   968→          data: {
   969→   969→            ideaId: id,
   970→   970→            code: idToCode.get(d.id)!,
   971→   971→            title: d.title,
   972→   972→            titleAr: d.titleAr,
   973→   973→            description: d.description,
   974→   974→            phase: d.phase,
   975→   975→            priority: d.priority,
   976→   976→            dependsOn: JSON.stringify(d.dependsOn.map((dep) => idToCode.get(dep)).filter((c): c is string => !!c)),
   977→   977→            acceptanceCriteria: JSON.stringify(d.acceptanceCriteria),
   978→   978→            deliverable: d.deliverable,
   979→   979→            requirementCodes: JSON.stringify(d.requirementCodes),
   980→   980→            estimate: d.estimate,
   981→   981→            status: 'confirmed',
   982→   982→            order: existingCount + n,
   983→   983→          },
   984→   984→        })
   985→   985→        n++
   986→   986→      }
   987→   987→      await logDecision(
   988→   988→        id,
   989→   989→        'Express: coverage tasks added',
   990→   990→        `${n} task(s) drafted and confirmed by delegation to cover ${uncovered.map((u) => u.code).join(', ')}`,
   991→   991→        'readiness'
   992→   992→      )
   993→   993→      return running(
   994→   994→        `Drafted ${n} task(s) to cover the newly confirmed requirements (${uncovered.map((u) => u.code).join(', ')})`,
   995→   995→        'Checking the readiness gate'
   996→   996→      )
   997→   997→    }
   998→   998→
   999→   999→    const report = await computeReadiness(id)
  1000→  1000→    if (!report.lockable) {
  1001→  1001→      return blocked(
  1002→  1002→        'The readiness gate still has hard blockers. Fix them in the readiness stage, then re-run express.',
  1003→  1003→        report.blockers
  1004→  1004→      )
  1005→  1005→    }
  1006→  1006→
  1007→  1007→    await db.idea.update({ where: { id }, data: { stage: 'package' } })
  1008→  1008→    await logDecision(id, 'Express: readiness gate passed — package locked', `Readiness score ${report.score}%`, 'readiness')
  1009→  1009→    return running(`Readiness gate passed (${report.score}%) — package locked`, 'Done')
  1010→  1010→  }
  1011→  1011→
  1012→  1012→  // ===== Stage: package =====
  1013→  1013→  return { status: 'done', step: 'Engineering package ready' }
  1014→  1014→}
