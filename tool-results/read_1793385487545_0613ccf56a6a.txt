    21→// ============================================================
    22→// Express autopilot — the client's explicit DELEGATION of the
    23→// remaining workflow to the virtual engineer. One unit of work
    24→// per call; the UI drives the loop and shows live progress.
    25→//
    26→// Philosophy preserved:
    27→// - Delegation is itself an explicit, logged act (never silence).
    28→// - Every auto-decision is written to the decision history with
    29→//   an Express marker and can be changed afterwards — the
    30→//   package can be reopened, artifacts edited, nothing restarts.
    31→// - The autopilot STOPS on genuine human decisions: critical
    32→//   consistency issues, readiness hard blockers, missing input.
    33→// ============================================================
    34→
    35→export interface ExpressStepResult {
    36→  status: 'running' | 'done' | 'blocked'
    37→  step: string
    38→  next?: string
    39→  reason?: string
    40→  blockers?: string[]
    41→}
    42→
    43→const running = (step: string, next?: string): ExpressStepResult => ({ status: 'running', step, next })
    44→const blocked = (reason: string, blockers?: string[]): ExpressStepResult => ({
    45→  status: 'blocked',
    46→  step: '',
    47→  reason,
    48→  blockers,
    49→})

    53→function safeArr(raw: string | null): string[] {
    54→  try {
    55→    const v = raw ? JSON.parse(raw) : []
    56→    return Array.isArray(v) ? v : []
    57→  } catch {
    58→    return []
    59→  }
    60→}

   102→const AUTO_ANSWER_PROMPT = `You are the discovery engine of an AI software engineer platform, running in EXPRESS mode: the client delegated the interview to you.
   103→For every pending question, pick the most sensible option for a lean, launchable MVP and justify it in one short sentence.
   104→

   108→- Prefer the simplest option that unblocks building; avoid options whose consequence implies heavy operations, cost or compliance unless the question clearly demands it.

   113→async function autoAnswerQuestions(ideaId: string): Promise<number> {
   114→  const pending = await db.question.findMany({
   115→    where: { ideaId, status: 'pending' },
   116→    orderBy: [{ order: 'asc' }, { createdAt: 'asc' }],
   117→  })
   118→  if (pending.length === 0) return 0
   119→
   120→  const idea = await db.idea.findUnique({ where: { id: ideaId } })
   121→  const interpretation = parseInterpretation(idea?.interpretation ?? null)
   122→  const ctx: ProjectContext | null = await buildProjectContext(ideaId)
   123→
   124→  const lines = pending.map((q, i) => {
   125→    const options = safeArr(q.options).length > 0 ? safeArr(q.options) : []
   126→    const optStr = options
   127→      .map((o) => {
   128→        const m = o as { label?: string; consequence?: string }
   129→        return `  - "${m.label ?? ''}"${m.consequence ? ` (consequence: ${m.consequence})` : ''}`
   130→      })
   131→      .join('\n')
   132→    return `Q${i + 1} [${q.area}] ${q.questionEn}\nwhy it matters: ${q.whyItMatters ?? '-'}\noptions:\n${optStr || '  (no predefined options — answer freely, max 12 words)'}`
   133→  })
   134→
   135→  const answered = await answeredItems(ideaId)
   136→  const answeredBlock =
   137→    answered.length > 0
   138→      ? `\nAlready answered by the client earlier (respect these as facts):\n${answered.map((a) => `- ${a.questionEn} → ${a.answer}`).join('\n')}`
   139→      : ''
   140→
   141→  const raw = await chatJson<{ answers?: Array<{ n?: number; choice?: string; why?: string }> }>(
   142→    AUTO_ANSWER_PROMPT,
   143→    `${ctx ? '' : ''}${interpretation ? `Product interpretation:\n${interpretation.englishInterpretation}\nProblem: ${interpretation.problemStatement}\n\n` : ''}Pending questions:\n${lines.join('\n\n')}${answeredBlock}`
   144→  )
   145→
   146→  const answers = Array.isArray(raw?.answers) ? raw.answers : []
   147→  let count = 0
   148→  for (let i = 0; i < pending.length; i++) {
   149→    const match = answers.find((a) => Number(a.n) === i + 1)
   150→    const fallbackOptions = safeArr(pending[i].options)
   151→    const fallbackLabel =
   152→      (fallbackOptions[0] as { label?: string } | undefined)?.label ?? 'Decided by the engineer (express)'
   153→    const answer = typeof match?.choice === 'string' && match.choice.trim() ? match.choice.trim() : fallbackLabel
   154→    await db.question.update({
   155→      where: { id: pending[i].id },
   156→      data: { answer, answerKind: 'choice', status: 'answered' },
   157→    })
   158→    // Every auto-answer is recorded as an OPEN assumption — never a silent fact.
   159→    await db.assumption.create({
   160→      data: {
   161→        ideaId,
   162→        text: `Express auto-answer — Q: ${pending[i].questionEn} → A: ${answer}`,
   163→        kind: 'suggested',
   164→        status: 'open',
   165→        source: 'express',
   166→      },
   167→    })
   168→    count++
   169→  }
   170→  return count
   171→}

   175→const AUTOFIX_PROMPT = `You are the consistency auto-fix engine of an AI software engineer platform, running in EXPRESS mode: the client delegated the fixes to you.
   176→For every critical consistency issue, propose the MINIMAL fix that makes the artifacts consistent. Prefer ADDING a missing field over removing a collected one (losing user data is worse than storing it).
   177→
   178→Allowed operations (use ONLY these):
   179→- {"op": "add_entity_field", "entity": "<entity name or code>", "field": {"name": "...", "type": "string", "required": true, "unique": false, "description": "..."}}
   180→- {"op": "remove_entity_field", "entity": "<entity name or code>", "field": "<field name>"}
   181→- {"op": "add_api_request_field", "api": "<api code like API-01>", "field": {"field": "...", "type": "string", "required": true, "notes": "..."}}
   182→- {"op": "remove_api_request_field", "api": "<api code>", "field": "<field name>"}
   183→
   184→If an issue cannot be fixed with these operations, return no fix for it (it will be escalated to the client).
   185→
   186→Return STRICT JSON only:
   187→{"fixes": [{"op": "...", "entity": "...", "api": "...", "field": {}, "issueIndex": 0, "note": "one sentence"}]}`
   188→
   189→async function autoFixCriticalIssues(
   190→  ideaId: string,
   191→  issues: Array<{ id: string; description: string; affected: string | null }>
   192→): Promise<number> {
   193→  const [entities, apis] = await Promise.all([
   194→    db.dataEntity.findMany({ where: { ideaId } }),
   195→    db.apiEndpoint.findMany({ where: { ideaId } }),
   196→  ])
   197→
   198→  const issueBlock = issues
   199→    .map((i, idx) => `Issue #${idx}: ${i.description}${i.affected ? ` (affected: ${i.affected})` : ''}`)
   200→    .join('\n')
   201→  const entityBlock = entities
   202→    .map((e) => `- ${e.code} "${e.name}": fields = ${e.fields}`)
   203→    .join('\n')
   204→  const apiBlock = apis
   205→    .map((a) => `- ${a.code} ${a.method} ${a.path}: request = ${a.request ?? '[]'}`)
   206→    .join('\n')
   207→
   208→  const raw = await chatJson<{ fixes?: Array<Record<string, unknown>> }>(
   209→    AUTOFIX_PROMPT,
   210→    `Critical issues:\n${issueBlock}\n\nCurrent data entities:\n${entityBlock}\n\nCurrent API endpoints:\n${apiBlock}`
   211→  )
   212→
   213→  const fixes = Array.isArray(raw?.fixes) ? raw.fixes : []
   214→  const parseJsonArr = (s: string | null): Array<Record<string, unknown>> => {
   215→    try {
   216→      const v = s ? JSON.parse(s) : []
   217→      return Array.isArray(v) ? v : []
   218→    } catch {
   219→      return []
   220→    }
   221→  }
   222→
   223→  let applied = 0
   224→  const appliedNotes: string[] = []
   225→  const norm = (s: string) => s.trim().toLowerCase()
   226→
   227→  for (const f of fixes) {
   228→    const op = typeof f.op === 'string' ? f.op : ''
   229→    const note = typeof f.note === 'string' ? f.note : ''
   230→    const issueIndex = typeof f.issueIndex === 'number' ? f.issueIndex : -1
   231→    const issue = issueIndex >= 0 && issueIndex < issues.length ? issues[issueIndex] : null
   232→
   233→    try {
   234→      if (op === 'add_entity_field' || op === 'remove_entity_field') {
   235→        const ref = String(f.entity ?? '')
   236→        const target = entities.find((e) => norm(e.name) === norm(ref) || norm(e.code) === norm(ref))
   237→        if (!target) continue
   238→        const fields = parseJsonArr(target.fields)
   239→        let next: Array<Record<string, unknown>>
   240→        if (op === 'add_entity_field') {
   241→          const fm = (f.field ?? {}) as Record<string, unknown>
   242→          const name = typeof fm.name === 'string' ? fm.name.trim() : ''
   243→          if (!name) continue
   244→          if (fields.some((x) => x.name === name)) {
   245→            // The data is already consistent — the issue row is stale.
   246→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   247→            applied++
   248→            appliedNotes.push(`${target.code}: field "${name}" already present — stale issue resolved`)
   249→            continue
   250→          }
   251→          next = [...fields, {
   252→            name,
   253→            type: typeof fm.type === 'string' && fm.type.trim() ? fm.type.trim() : 'string',
   254→            required: fm.required === true,
   255→            unique: fm.unique === true,
   256→            description: typeof fm.description === 'string' ? fm.description : '',
   257→          }]
   258→        } else {
   259→          const fname = String(f.field ?? '').trim()
   260→          next = fields.filter((x) => x.name !== fname)
   261→          if (next.length === fields.length) {
   262→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   263→            applied++
   264→            appliedNotes.push(`${target.code}: field "${fname}" already absent — stale issue resolved`)
   265→            continue
   266→          }
   267→        }
   268→        await db.dataEntity.update({
   269→          where: { id: target.id },
   270→          data: {
   271→            fields: JSON.stringify(next),
   272→            status: target.status === 'confirmed' ? 'confirmed' : target.status,
   273→          },
   274→        })
   275→        applied++
   276→        appliedNotes.push(`${target.code}: ${op === 'add_entity_field' ? 'added' : 'removed'} field — ${note}`)
   277→        if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   278→      } else if (op === 'add_api_request_field' || op === 'remove_api_request_field') {
   279→        const ref = String(f.api ?? '')
   280→        const target = apis.find(
   281→          (a) => norm(a.code) === norm(ref) || norm(a.path) === norm(ref) || norm(`${a.method} ${a.path}`) === norm(ref)
   282→        )
   283→        if (!target) continue
   284→        const request = parseJsonArr(target.request)
   285→        let next: Array<Record<string, unknown>>
   286→        if (op === 'add_api_request_field') {
   287→          const fm = (f.field ?? {}) as Record<string, unknown>
   288→          const field = typeof fm.field === 'string' ? fm.field.trim() : ''
   289→          if (!field) continue
   290→          if (request.some((x) => x.field === field)) {
   291→            // Already consistent — the issue row is stale.
   292→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   293→            applied++
   294→            appliedNotes.push(`${target.code}: request field "${field}" already present — stale issue resolved`)
   295→            continue
   296→          }
   297→          next = [...request, {
   298→            field,
   299→            type: typeof fm.type === 'string' && fm.type.trim() ? fm.type.trim() : 'string',
   300→            required: fm.required === true,
   301→            notes: typeof fm.notes === 'string' ? fm.notes : '',
   302→          }]
   303→        } else {
   304→          const fname = String(f.field ?? '').trim()
   305→          next = request.filter((x) => x.field !== fname)
   306→          if (next.length === request.length) {
   307→            if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   308→            applied++
   309→            appliedNotes.push(`${target.code}: request field "${fname}" already absent — stale issue resolved`)
   310→            continue
   311→          }
   312→        }
   313→        await db.apiEndpoint.update({
   314→          where: { id: target.id },
   315→          data: {
   316→            request: JSON.stringify(next),
   317→            status: target.status === 'confirmed' ? 'confirmed' : target.status,
   318→          },
   319→        })
   320→        applied++
   321→        appliedNotes.push(`${target.code}: ${op === 'add_api_request_field' ? 'added' : 'removed'} request field — ${note}`)
   322→        if (issue) await db.consistencyIssue.update({ where: { id: issue.id }, data: { status: 'resolved' } })
   323→      }
   324→    } catch {
   325→      // a single malformed fix must not abort the others
   326→    }
   327→  }
   328→
   329→  if (applied > 0) {
   330→    await logDecision(
   331→      ideaId,
   332→      'Express: critical issues auto-fixed',
   333→      appliedNotes.slice(0, 5).join(' | '),
   334→      'readiness'
   335→    )
   336→  }
   337→  return applied
   338→}

   340→const RESOLVE_MUST_PROMPT = `You are the requirements engineer of an AI software engineer platform, running in EXPRESS mode: the client delegated the final call on open must-have requirements.
   341→For each unresolved must-priority requirement, decide confirm or reject for the lean, launchable MVP.
   342→
   343→Rules:
   344→- CONFIRM when the requirement is needed by the accepted feature baseline and does not contradict the confirmed set or the architecture.
   345→- REJECT when it contradicts confirmed requirements, duplicates one, or is not needed for the MVP.
   346→- Never invent new requirements; only decide the listed ones.
   347→
   348→Return STRICT JSON only:
   349→{"decisions": [{"code": "FR-001", "decision": "confirmed", "why": "one sentence"}]}`
   350→
   351→async function resolveUnresolvedMusts(ideaId: string): Promise<number> {
   352→  const rows = await db.requirement.findMany({ where: { ideaId, status: 'unresolved', priority: 'must' } })
   353→  if (rows.length === 0) return 0
   354→
   355→  const confirmed = await db.requirement.findMany({
   356→    where: { ideaId, status: 'confirmed' },
   357→    select: { code: true, title: true },
   358→  })
   359→
   360→  let decisions: Array<{ code?: string; decision?: string; why?: string }> = []
   361→  try {
   362→    const raw = await chatJson<{ decisions?: Array<{ code?: string; decision?: string; why?: string }> }>(
   363→      RESOLVE_MUST_PROMPT,
   364→      `Unresolved must-have requirements:
   365→${rows.map((r) => `- ${r.code}: ${r.title}\n  ${r.description}\n  evidence: ${r.evidence ?? '-'}`).join('\n')}
   366→
   367→Confirmed requirements (facts):
   368→${confirmed.map((c) => `- ${c.code}: ${c.title}`).join('\n')}`
   369→    )
   370→    decisions = Array.isArray(raw?.decisions) ? raw.decisions : []
   371→  } catch {
   372→    decisions = []
   373→  }
   374→
   375→  let resolved = 0
   376→  for (const r of rows) {
   377→    const d = decisions.find((x) => x.code === r.code)
   378→    const decision = d?.decision === 'rejected' ? 'rejected' : 'confirmed'
   379→    const why = typeof d?.why === 'string' ? d.why : 'delegated fallback — confirmed to keep the must-have scope explicit'
   380→    await db.requirement.update({ where: { id: r.id }, data: { status: decision } })
   381→    await logDecision(ideaId, `Express: must-have ${r.code} → ${decision}`, why, 'readiness')
   382→    resolved++
   383→  }
   384→  return resolved
   385→}

   389→export async function runExpressStep(ideaId: string): Promise<ExpressStepResult> {
   390→  const idea = await db.idea.findUnique({ where: { id: ideaId } })
   391→  if (!idea) return blocked('Idea not found.')
   392→  const id = ideaId
   393→
   394→  // ===== Stage: understanding =====
   395→  if (idea.stage === 'understanding') {
   396→    const interpretation = parseInterpretation(idea.interpretation)
   397→    if (!interpretation) {
   398→      return blocked('There is no interpretation yet — describe the idea and run the understanding step first.')
   399→    }
   400→    const drafts = await generateQuestions({ interpretation, answered: [], uncoveredAreas: [...AREAS], weakAreas: [] })
   401→    let order = 0
   402→    for (const draft of drafts) {
   403→      await db.question.create({
   404→        data: {
   405→          ideaId: id,
   406→          area: draft.area,
   407→          questionEn: draft.questionEn,
   408→          questionAr: draft.questionAr,
   409→          whyItMatters: draft.whyItMatters,
   410→          options: JSON.stringify(draft.options),
   411→          status: 'pending',
   412→          order: order++,
   413→        },
   414→      })
   415→    }
   416→    await db.idea.update({ where: { id }, data: { stage: 'discovery' } })
   417→    await logDecision(id, 'Express: interpretation approved — discovery started', `${drafts.length} questions generated`, 'understanding')
   418→    return running('Interpretation approved — discovery questions generated', 'Auto-answering the interview questions')
   419→  }
   420→
   421→  // ===== Stage: discovery =====
   422→  if (idea.stage === 'discovery') {
   423→    const pending = await db.question.count({ where: { ideaId: id, status: 'pending' } })
   424→    if (pending > 0) {
   425→      const n = await autoAnswerQuestions(id)
   426→      await logDecision(id, 'Express: interview auto-answered', `${n} answers chosen by the engineer — every answer recorded as an open assumption`, 'discovery')
   427→      return running(`Auto-answered ${n} interview questions (each recorded as an open assumption)`, 'Compiling the discovery summary')
   428→    }
   429→
   430→    const answered = await db.question.count({ where: { ideaId: id, status: 'answered' } })
   431→    if (answered < 5) {
   432→      return blocked('Not enough answered questions to compile a discovery summary (minimum 5).')
   433→    }
   434→
   435→    const interpretation = parseInterpretation(idea.interpretation)
   436→    if (!interpretation) return blocked('No interpretation available.')
   437→    const summary = await compileDiscoverySummary({ interpretation, answered: await answeredItems(id) })
   438→    for (const risk of summary.openRisks) {
   439→      await db.assumption.create({ data: { ideaId: id, text: risk, kind: 'uncertain', status: 'open', source: 'discovery' } })
   440→    }
   441→    for (const s of summary.suggestedAssumptions) {
   442→      await db.assumption.create({ data: { ideaId: id, text: s, kind: 'suggested', status: 'open', source: 'discovery' } })
   443→    }
   444→    await db.idea.update({ where: { id }, data: { stage: 'features', summary: JSON.stringify(summary) } })
   445→    await logDecision(id, 'Express: discovery finished — summary compiled', `${answered} questions answered`, 'discovery')
   446→    return running('Discovery summary compiled', 'Generating the feature inventory')
   447→  }
   448→
   449→  // ===== Stage: features =====
   450→  if (idea.stage === 'features') {
   451→    const features = await db.feature.findMany({ where: { ideaId: id } })
   452→    if (features.length === 0) {
   453→      const interpretation = parseInterpretation(idea.interpretation)
   454→      if (!interpretation) return blocked('No interpretation available.')
   455→      const inventory = await generateFeatureInventory({ interpretation, answered: await answeredItems(id) })
   456→      let order = 0
   457→      for (const f of inventory.features) {
   458→        await db.feature.create({
   459→          data: {
   460→            ideaId: id,
   461→            module: f.module,
   462→            feature: f.feature,
   463→            user: f.user,
   464→            priority: f.priority,
   465→            status: f.status,
   466→            rationale: f.rationale,
   467→            origin: 'ai',
   468→            order: order++,
   469→          },
   470→        })
   471→      }
   472→      await db.idea.update({
   473→        where: { id },
   474→        data: {
   475→          featuresMeta: JSON.stringify({
   476→            missingFeatures: inventory.missingFeatures,
   477→            dependencies: inventory.dependencies,
   478→            releaseNote: inventory.releaseNote,
   479→          }),
   480→        },
   481→      })
   482→      await logDecision(id, 'Express: feature inventory generated', `${inventory.features.length} features proposed`, 'features')
   483→      return running(`Feature inventory generated (${inventory.features.length} features)`, 'Accepting the feature baseline')
   484→    }
   485→
   486→    const unreviewed = features.filter((f) => f.status === 'proposed' || f.status === 'needs_confirmation')
   487→    if (unreviewed.length > 0) {
   488→      await db.feature.updateMany({
   489→        where: { ideaId: id, status: { in: ['proposed', 'needs_confirmation'] } },
   490→        data: { status: 'accepted' },
   491→      })

   957→        return blocked(
   958→          'These confirmed requirements have no implementation task and the engineer could not draft one automatically:',
   959→          uncovered.map((u) => `${u.code}: ${u.title}`)
   960→        )
   961→      }
   962→      const existingCount = await db.implTask.count({ where: { ideaId: id } })
   963→      let n = 0
   964→      const idToCode = new Map<string, string>()
   965→      for (const d of targeted) idToCode.set(d.id, `T-${String(existingCount + idToCode.size + 1).padStart(2, '0')}`)
   966→      for (const d of targeted) {
   967→        await db.implTask.create({
   968→          data: {
   969→            ideaId: id,
   970→            code: idToCode.get(d.id)!,
   971→            title: d.title,
   972→            titleAr: d.titleAr,
   973→            description: d.description,
   974→            phase: d.phase,
   975→            priority: d.priority,
   976→            dependsOn: JSON.stringify(d.dependsOn.map((dep) => idToCode.get(dep)).filter((c): c is string => !!c)),
   977→            acceptanceCriteria: JSON.stringify(d.acceptanceCriteria),
   978→            deliverable: d.deliverable,
   979→            requirementCodes: JSON.stringify(d.requirementCodes),
   980→            estimate: d.estimate,
   981→            status: 'confirmed',
   982→            order: existingCount + n,
   983→          },
   984→        })
   985→        n++
   986→      }
   987→      await logDecision(
   988→        id,
   989→        'Express: coverage tasks added',
   990→        `${n} task(s) drafted and confirmed by delegation to cover ${uncovered.map((u) => u.code).join(', ')}`,
   991→        'readiness'
   992→      )
   993→      return running(
   994→        `Drafted ${n} task(s) to cover the newly confirmed requirements (${uncovered.map((u) => u.code).join(', ')})`,
   995→        'Checking the readiness gate'
   996→      )
   997→    }
   998→
   999→    const report = await computeReadiness(id)
  1000→    if (!report.lockable) {
  1001→      return blocked(
  1002→        'The readiness gate still has hard blockers. Fix them in the readiness stage, then re-run express.',
  1003→        report.blockers
  1004→      )
  1005→    }
  1006→
  1007→    await db.idea.update({ where: { id }, data: { stage: 'package' } })
  1008→    await logDecision(id, 'Express: readiness gate passed — package locked', `Readiness score ${report.score}%`, 'readiness')
  1009→    return running(`Readiness gate passed (${report.score}%) — package locked`, 'Done')
  1010→  }
  1011→
  1012→  // ===== Stage: package =====
  1013→  return { status: 'done', step: 'Engineering package ready' }
  1014→}
