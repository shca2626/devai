    14→function esc(v: string | null | undefined): string {
    15→  return (v ?? '').replace(/\|/g, '\\|').replace(/\n/g, ' ')
    16→}
    17→
    18→function parse<T>(raw: string | null | undefined, fallback: T): T {
    19→  if (!raw) return fallback
    20→  try {
    21→    return JSON.parse(raw) as T
    22→  } catch {
    23→    return fallback
    24→  }
    25→}

    27→function mermaidErd(entities: Array<{ name: string; relationships: string | null }>): string {
    28→  const rels = entities.flatMap((e) =>
    29→    parse<Array<{ target: string; type: string }>>(e.relationships, []).map((r) => ({ from: e.name, ...r }))
    30→  )
    31→  const sanitize = (n: string) => n.replace(/[^a-zA-Z0-9]/g, '_')
    32→  const lines = ['erDiagram']
    33→  for (const e of entities) {
    34→    lines.push(`  ${sanitize(e.name)} {`)
    35→    lines.push('    string id PK')
    36→    lines.push('  }')
    37→  }
    38→  for (const r of rels) {
    39→    const cardinality =
    40→      r.type.includes('many-to-many')
    41→        ? '}o..o{'
    42→        : r.type.includes('one-to-many')
    43→          ? '||--o{'
    44→          : r.type.includes('one-to-one')
    45→            ? '||--||'
    46→            : '||--o{'
    47→    lines.push(`  ${sanitize(r.from)} ${cardinality} ${sanitize(r.target)} : "${r.type}"`)
    48→  }
    49→  return lines.join('\n')
    50→}

   489→      const tech = parse<Array<{ area: string; choice: string; rationale: string; tradeOffs: string; simplestOption: boolean }>>(arch.techStack, [])
   490→      if (tech.length) {
   491→        L.push('| Area | Recommendation | Rationale | Trade-offs | Simplest option |')
   492→        L.push('| --- | --- | --- | --- | --- |')
   493→        for (const t of tech) {
   494→          L.push(`| ${esc(t.area)} | ${esc(t.choice)} | ${esc(t.rationale)} | ${esc(t.tradeOffs)} | ${t.simplestOption ? 'yes' : 'no'} |`)
   495→        }
   496→        L.push('')
   497→      }
   498→      const integrations = parse<Array<{ system: string; purpose: string; necessity: string }>>(arch.integrations, [])
   499→      if (integrations.length) {
   500→        L.push(`**Integrations:** ${integrations.map((i) => `${i.system} — ${i.purpose} (${i.necessity})`).join(' · ')}`)
   501→        L.push('')
   502→      }
   503→      if (arch.notes) {
   504→        L.push(`**Notes:** ${arch.notes}`)
   505→        L.push('')
   506→      }
   507→    }
   508→    if (idea.tradeStudy && idea.tradeStudy.status === 'confirmed') {
   509→      const ts = idea.tradeStudy
   510→      L.push('### Architecture trade-off study')
   511→      L.push('')
   512→      L.push(`**Recommended: ${ts.recommended}**${ts.rationale ? ` — ${ts.rationale}` : ''}`)
   513→      L.push('')
   514→      const candidates = parse<Array<{ name: string; style: string; summary: string; pros: string[]; cons: string[]; cost: string; complexity: string; teamFit: string; risks: string[]; fit: string }>>(ts.candidates, [])
   515→      L.push('| Candidate | Style | Cost | Complexity | Pros | Cons |')
   516→      L.push('| --- | --- | --- | --- | --- | --- |')
   517→      for (const c of candidates) {
   518→        L.push(`| ${c.name === ts.recommended ? `**${esc(c.name)}**` : esc(c.name)} | ${esc(c.style)} | ${esc(c.cost)} | ${esc(c.complexity)} | ${c.pros.map(esc).join(' · ')} | ${c.cons.map(esc).join(' · ')} |`)
   519→      }
   520→      L.push('')
   521→    }
   522→    if (confirmedAdrs.length) {
   523→      L.push('### Architecture decision records (ADRs)')
   524→      L.push('')
   525→      for (const a of confirmedAdrs) {
   526→        L.push(`#### ${a.code} — ${a.title}`)
   527→        L.push('')
   528→        L.push(`**Context.** ${a.context}`)
   529→        L.push('')
   530→        L.push(`**Decision.** ${a.decision}`)
   531→        L.push('')
   532→        const alts = parse<Array<{ option: string; whyNot: string }>>(a.alternatives, [])
   533→        if (alts.length) {
   534→          L.push(`**Alternatives rejected:** ${alts.map((x) => `${x.option} (${x.whyNot})`).join(' · ')}`)
   535→          L.push('')
   536→        }
   537→        const conseq = parse<{ positive?: string[]; negative?: string[]; risks?: string[] }>(a.consequences, {})
   538→        const conseqParts: string[] = []
   539→        if (conseq.positive?.length) conseqParts.push(`positive: ${conseq.positive.join(' · ')}`)
   540→        if (conseq.negative?.length) conseqParts.push(`negative: ${conseq.negative.join(' · ')}`)
   541→        if (conseq.risks?.length) conseqParts.push(`risks: ${conseq.risks.join(' · ')}`)
   542→        if (conseqParts.length) {
   543→          L.push(`**Consequences.** ${conseqParts.join(' — ')}`)
   544→          L.push('')
   545→        }
   546→        const linked = parse<string[]>(a.linkedRequirements, [])
   547→        if (linked.length) L.push(`**Serves:** ${linked.join(', ')}`)
   548→        L.push('')
   549→      }
   550→    }
   551→
   552→    // 04 — UI/UX Specification
   553→    L.push('## 04 — UI/UX Specification')
   554→    L.push('')
   555→    for (const s of confirmedScreens) {
   556→      L.push(`#### ${s.code} — ${s.name}${s.nameAr ? ` (${s.nameAr})` : ''}`)
   557→      L.push('')
   558→      if (s.route) L.push(`**Route:** \`${s.route}\` · **Roles:** ${parse<string[]>(s.roles, []).join(', ') || 'all'}`)
   559→      if (s.purpose) L.push(`**Purpose:** ${s.purpose}`)
   560→      L.push('')
   561→      const elements = parse<Array<{ element: string; behavior: string }>>(s.elements, [])
   562→      if (elements.length) {
   563→        L.push('| Element | Behavior |')
   564→        L.push('| --- | --- |')
   565→        for (const el of elements) L.push(`| ${esc(el.element)} | ${esc(el.behavior)} |`)
   566→        L.push('')
   567→      }
   568→      const states = parse<{ loading?: string; empty?: string; error?: string }>(s.states, {})
   569→      L.push(`**States** — Loading: ${states.loading ?? '—'} · Empty: ${states.empty ?? '—'} · Error: ${states.error ?? '—'}`)
   570→      L.push('')
   571→      if (s.rtlNote) L.push(`**RTL:** ${s.rtlNote}`)
   572→      const linkedReqs = parse<string[]>(s.linkedRequirements, [])
   573→      if (linkedReqs.length) L.push(`**Implements:** ${linkedReqs.join(', ')}`)
   574→      L.push('')
   575→    }
   576→
   577→    // 05 — Quality & Operations
   578→    L.push('## 05 — Quality & Operations')
   579→    L.push('')
   580→    L.push('### Non-functional requirements')
   581→    L.push('')
   582→    L.push('| Code | Requirement | Category | Acceptance criteria |')
   583→    L.push('| --- | --- | --- | --- |')
   584→    for (const r of nfrs) {
   585→      L.push(`| ${r.code} | ${esc(r.title)} | ${esc(r.category)} | ${esc(parse<string[]>(r.acceptanceCriteria, []).join(' · '))} |`)
   586→    }
   587→    L.push('')
   588→    L.push('### Risk register (with mitigation decisions)')
   589→    L.push('')
   590→    L.push('| Code | Category | Risk | Likelihood | Impact | Mitigation | Decision | Basis |')
   591→    L.push('| --- | --- | --- | --- | --- | --- | --- | --- |')
   592→    for (const r of reviewedRisks) {
   593→      L.push(`| ${r.code} | ${r.category} | ${esc(r.description)} | ${r.likelihood} | ${r.impact} | ${esc(r.mitigation)} | ${r.status} | ${r.basis === 'verified_fact' ? 'fact' : 'judgment'} |`)
   594→    }
   595→    L.push('')
   596→    L.push('### Verification strategy')
   597→    L.push('')
   598→    L.push('Every confirmed requirement ships with testable acceptance criteria; the implementation is complete only when each passes:')
   599→    L.push('')
   600→    for (const r of idea.requirements.filter((x) => x.status === 'confirmed')) {
   601→      L.push(`- **${r.code} — ${r.title}**`)
   602→      for (const c of parse<string[]>(r.acceptanceCriteria, [])) L.push(`  - ${c}`)
   603→    }
   604→    L.push('')
   605→    if (confirmedTestCases.length > 0) {
   606→      L.push('### Test suite (confirmed cases)')
   607→      L.push('')
   608→      if (idea.testCoverage && idea.testCoverage.status === 'confirmed') {
   609→        L.push(`**Acceptance-criteria coverage: ${idea.testCoverage.score}/100** (computed by the platform from the cases below).`)
   610→        L.push('')
   611→      }
   612→      L.push('| Code | Case | Level | Kind | Requirement | Criterion | Priority | Expected |')
   613→      L.push('| --- | --- | --- | --- | --- | --- | --- | --- |')
   614→      for (const c of confirmedTestCases) {
   615→        L.push(
   616→          `| ${c.code} | ${esc(c.title)} | ${c.level} | ${c.kind} | ${c.linkedRequirement} | ${c.linkedCriterion ? `#${c.linkedCriterion}` : '—'} | ${c.priority} | ${esc(c.expected)} |`
   617→        )
   618→      }
   619→      L.push('')
   620→      for (const c of confirmedTestCases) {
   621→        const steps = parse<string[]>(c.steps, [])
   622→        const pre = parse<string[]>(c.preconditions, [])
   623→        if (steps.length || pre.length) {
   624→          L.push(`#### ${c.code} — ${esc(c.title)}`)
   625→          L.push('')
   626→          if (pre.length) {
   627→            L.push(`**Preconditions:** ${pre.join(' · ')}`)
   628→            L.push('')
   629→          }
   630→          if (steps.length) {
   631→            steps.forEach((s, i) => L.push(`${i + 1}. ${s}`))
   632→            L.push('')
   633→          }
   634→          L.push(`**Expected:** ${esc(c.expected)}`)
   635→          L.push('')
   636→        }
   637→      }
   638→    }
   639→    if (idea.edgeCases && idea.edgeCases.status === 'confirmed') {
   640→      L.push('### Edge case register')
   641→      L.push('')
   642→      if (idea.edgeCases.note) {
   643→        L.push(`> ${esc(idea.edgeCases.note)}`)
   644→        L.push('')
   645→      }
   646→      const groups = parse<Array<{ feature: string; cases: Array<{ scenario: string; category: string; risk: string; howToTest: string; linkedRequirement: string | null }> }>>(idea.edgeCases.groups, [])
   647→      for (const g of groups) {
   648→        L.push(`#### ${esc(g.feature)}`)
   649→        L.push('')
   650→        L.push('| Category | Risk | Scenario | How to test | Req |')
   651→        L.push('| --- | --- | --- | --- | --- |')
   652→        for (const c of g.cases) {
   653→          L.push(`| ${esc(c.category.replace(/_/g, ' '))} | ${c.risk} | ${esc(c.scenario)} | ${esc(c.howToTest)} | ${c.linkedRequirement ?? '—'} |`)
   654→        }
   655→        L.push('')
   656→      }
   657→    }
   658→    if (confirmedE2e.length > 0) {
   659→      L.push('### E2E scenarios')
   660→      L.push('')
   661→      for (const s of confirmedE2e) {
   662→        L.push(`#### ${s.code} — ${esc(s.title)}${s.priority === 'smoke' ? ' (smoke)' : ''}`)
   663→        L.push('')
   664→        L.push(`**Persona:** ${esc(s.persona)}`)
   665→        L.push('')
   666→        const given = parse<string[]>(s.given, [])
   667→        const when = parse<string[]>(s.when, [])
   668→        const then = parse<string[]>(s.then, [])
   669→        const screens = parse<string[]>(s.screens, [])
   670→        const linked = parse<string[]>(s.linkedRequirements, [])
   671→        if (given.length) L.push(`- **Given** ${given.join(' · ')}`)
   672→        if (when.length) L.push(`- **When** ${when.join(' · ')}`)
   673→        if (then.length) L.push(`- **Then** ${then.join(' · ')}`)
   674→        if (screens.length) L.push(`- **Screens** ${screens.join(' → ')}`)
   675→        if (linked.length) L.push(`- **Proves** ${linked.join(', ')}`)
   676→        L.push('')
   677→      }
   678→    }
   679→    if (idea.threatModel && idea.threatModel.status === 'confirmed') {
   680→      const tm = idea.threatModel
   681→      L.push('### STRIDE threat model')
   682→      L.push('')
   683→      if (tm.summary) {
   684→        L.push(`**Security posture.** ${tm.summary}`)
   685→        L.push('')
   686→      }
   687→      const findings = parse<Array<{ category: string; threat: string; component: string; attackPath: string; likelihood: string; impact: string; risk: string; mitigation: string }>>(tm.findings, [])
   688→      if (findings.length) {
   689→        L.push('| STRIDE | Threat | Component | Attack path | Risk | Mitigation |')
   690→        L.push('| --- | --- | --- | --- | --- | --- |')
   691→        for (const f of findings) {
   692→          L.push(`| ${f.category} | ${esc(f.threat)} | ${esc(f.component)} | ${esc(f.attackPath)} | ${f.likelihood}/${f.impact} → ${f.risk} | ${esc(f.mitigation)} |`)
   693→        }
   694→        L.push('')
   695→      }
   696→      if (tm.residual) {
   697→        L.push(`**Residual risk.** ${tm.residual}`)
   698→        L.push('')
   699→      }
   700→    }
   701→    if (idea.scaleReport && idea.scaleReport.status === 'confirmed') {
   702→      const sr = idea.scaleReport
   703→      L.push('### Scale simulation (10x / 100x)')
   704→      L.push('')
   705→      const assumptions = parse<Array<{ assumption: string; basis: string }>>(sr.assumptions, [])
   706→      if (assumptions.length) {

   717→        L.push('')
   718→      }
   719→      const scenarios = parse<Array<{ label: string; load: string; expected: string; breaks: string; actions: string[] }>>(sr.scenarios, [])

   725→        if (s.actions?.length) {
   726→          for (const [i, a] of s.actions.entries()) L.push(`${i + 1}. ${a}`)
   727→          L.push('')
   728→        }
   729→      }
   730→      if (sr.plan) {
   731→        L.push(`**Scaling plan.** ${sr.plan}`)
   732→        L.push('')
   733→      }
   734→    }
   735→

   738→    L.push('')
   739→    const phases = [...new Set(confirmedTasks.map((t) => t.phase))].sort((a, b) => a - b)
   740→    for (const phase of phases) {
   741→      L.push(`### Phase ${phase}`)
   742→      L.push('')
   743→      const tasks = confirmedTasks.filter((t) => t.phase === phase)
   744→      for (const t of tasks) {
   745→        const deps = parse<string[]>(t.dependsOn, [])

   747→        L.push('')
   748→        L.push(t.description)
   749→        L.push('')
   750→        L.push(`**Priority:** ${t.priority} · **Estimate:** ${t.estimate ?? 'M'} · **Deliverable:** ${t.deliverable ?? '—'}${deps.length ? ` · **Depends on:** ${deps.join(', ')}` : ''}`)
   751→        L.push('')
   752→        const criteria = parse<string[]>(t.acceptanceCriteria, [])
   753→        if (criteria.length) {
   754→          L.push('**Done when:**')
   755→          L.push('')
   756→          for (const c of criteria) L.push(`- ${c}`)
   757→          L.push('')
   758→        }
   759→        const reqs = parse<string[]>(t.requirementCodes, [])
   760→        if (reqs.length) L.push(`**Implements:** ${reqs.join(', ')}`)
   761→        L.push('')
   762→      }
   763→    }
   764→
   765→    // Appendix — traceability
   766→    L.push('## Appendix A — Requirements traceability matrix')
   767→    L.push('')
   768→    L.push('| Requirement | Processes | APIs | Screens | Tasks |')
   769→    L.push('| --- | --- | --- | --- | --- |')
   770→    for (const r of idea.requirements.filter((x) => x.status === 'confirmed')) {
   771→      L.push(
   772→        `| ${r.code} ${esc(r.title)} | ${(procLinks.get(r.code) ?? []).join(', ') || '—'} | ${(apiLinks.get(r.code) ?? []).join(', ') || '—'} | ${(screenLinks.get(r.code) ?? []).join(', ') || '—'} | ${(taskLinks.get(r.code) ?? []).join(', ') || '—'} |`
   773→      )
   774→    }
   775→    L.push('')
   776→
   777→    // Appendix — registers
   778→    L.push('## Appendix B — Assumption register')
   779→    L.push('')
   780→    if (idea.assumptions.length === 0) {
   781→      L.push('No assumptions were tracked for this project.')
   782→    } else {
   783→      L.push('| Assumption | Kind | Status |')
   784→      L.push('| --- | --- | --- |')
   785→      for (const a of idea.assumptions) L.push(`| ${esc(a.text)} | ${a.kind} | ${a.status} |`)
   786→    }
   787→    L.push('')
   788→    L.push('## Appendix C — Decision history (audit trail)')
   789→    L.push('')
   790→    for (const d of idea.decisions) {
   791→      L.push(`- **${d.action}** — ${d.createdAt.toISOString().slice(0, 16).replace('T', ' ')}${d.detail ? ` — _${d.detail}_` : ''}`)
   792→    }
   793→    L.push('')
   794→
   795→    return new Response(L.join('\n'), {
   796→      headers: {
   797→        'Content-Type': 'text/markdown; charset=utf-8',
   798→        'Content-Disposition': `attachment; filename="engineering-package-${slug}.md"`,
   799→      },
   800→    })
   801→  } catch (error) {
   802→    console.error('[api/ideas/[id]/package/export GET]', error)
   803→    return Response.json({ error: 'Failed to export the engineering package.' }, { status: 500 })
   804→  }
