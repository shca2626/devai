     1→'use client'
     2→
     3→import { useState } from 'react'
     4→import { Button } from '@/components/ui/button'
     5→import { Badge } from '@/components/ui/badge'
     6→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
     7→import { useApp } from '@/components/app/context'
     8→import { CodeChip, ReviewBadge, WorkingPanel } from '@/components/app/bits'
     9→import type { IdeaDetail } from '@/lib/types'
    10→import type { DictKey } from '@/lib/i18n'
    11→import {
    12→  Boxes,
    13→  Check,
    14→  ChevronDown,
    15→  ChevronUp,
    16→  Database,
    17→  GitBranch,
    18→  LayoutTemplate,
    19→  Loader2,
    20→  MessageSquareWarning,
    21→  Pencil,
    22→  Plus,
    23→  Server,
    24→  Workflow,
    25→  X,
    26→} from 'lucide-react'
    27→import { Input } from '@/components/ui/input'
    28→import { Checkbox } from '@/components/ui/checkbox'
    29→import { useToast } from '@/hooks/use-toast'
    30→
    31→// ============================================================
    32→// Stage 6 — System design: five engines (processes, data, API,
    33→// screens, architecture) turn confirmed requirements into the
    34→// system specification. Each artifact set is reviewed item by
    35→// item; rejected artifacts are never a dead end — they can be
    36→// regenerated, optionally with client feedback the engine must
    37→// address. Only confirmed decisions are protected.
    38→// ============================================================
    39→
    40→type Kind = 'processes' | 'entities' | 'apis' | 'screens' | 'architecture'
    41→
    42→const GENERATE_KEYS: Record<Kind, { label: DictKey; steps: [DictKey, DictKey] }> = {
    43→  processes: { label: 'generateProcesses', steps: ['genProcesses1', 'genProcesses2'] },
    44→  entities: { label: 'generateEntities', steps: ['genEntities1', 'genEntities2'] },
    45→  apis: { label: 'generateApis', steps: ['genApis1', 'genApis2'] },
    46→  screens: { label: 'generateScreens', steps: ['genScreens1', 'genScreens2'] },
    47→  architecture: { label: 'generateArchitecture', steps: ['genArch1', 'genArch2'] },
    48→}
    49→
    50→export function DesignStage({ detail }: { detail: IdeaDetail }) {
    51→  const { t, setDetail, refreshIdeas } = useApp()
    52→  const { toast } = useToast()
    53→  const { idea, processes, entities, apis, screens, architecture } = detail
    54→  // NO LIMITS: design artifacts can be generated, reviewed and approved at ANY stage
    55→  const active = true
    56→  const [busyKind, setBusyKind] = useState<Kind | null>(null)
    57→  const [busyGate, setBusyGate] = useState(false)
    58→  const [expanded, setExpanded] = useState<string | null>(null)
    59→  const [tab, setTab] = useState('processes')
    60→  const [feedback, setFeedback] = useState<Record<Kind, string>>({
    61→    processes: '',
    62→    entities: '',
    63→    apis: '',
    64→    screens: '',
    65→    architecture: '',
    66→  })
    67→  const [editingArch, setEditingArch] = useState(false)
    68→  const [draftStack, setDraftStack] = useState<Array<{ area: string; choice: string; rationale: string; tradeOffs: string; simplestOption: boolean }>>([])
    69→  const [draftNotes, setDraftNotes] = useState('')
    70→
    71→  async function generate(kind: Kind, kindFeedback?: string) {
    72→    setBusyKind(kind)
    73→    const res = await fetch(`/api/ideas/${idea.id}/design/generate`, {
    74→      method: 'POST',
    75→      headers: { 'Content-Type': 'application/json' },
    76→      body: JSON.stringify({ kind, feedback: kindFeedback }),
    77→    })
    78→    const data = await res.json()
    79→    setBusyKind(null)
    80→    if (!res.ok) {
    81→      toast({ description: data.error ?? t('aiError'), variant: 'destructive' })
    82→      return
    83→    }
    84→    setFeedback((f) => ({ ...f, [kind]: '' }))
    85→    setDetail(data as IdeaDetail)
    86→    refreshIdeas()
    87→  }
    88→
    89→  function startEditArch() {
    90→    if (!architecture) return
    91→    setDraftStack(architecture.techStack.map((s) => ({ ...s })))
    92→    setDraftNotes(architecture.notes ?? '')
    93→    setEditingArch(true)
    94→  }
    95→
    96→  async function saveArchEdits() {
    97→    if (!architecture) return
    98→    setBusyKind('architecture')
    99→    const res = await fetch(`/api/artifacts/architecture/${architecture.id}`, {
   100→      method: 'PATCH',
   101→      headers: { 'Content-Type': 'application/json' },
   102→      body: JSON.stringify({ edits: { techStack: draftStack, notes: draftNotes } }),
   103→    })
   104→    const data = await res.json()
   105→    setBusyKind(null)
   106→    if (!res.ok) {
   107→      toast({ description: data.error ?? t('aiError'), variant: 'destructive' })
   108→      return
   109→    }
   110→    setEditingArch(false)
   111→    setDetail(data as IdeaDetail)
   112→    refreshIdeas()
   113→    toast({ description: t('archEditedNeedsReconfirm') })
   114→  }
   115→
   116→  async function review(kind: string, id: string, status: string) {
   117→    const res = await fetch(`/api/artifacts/${kind}/${id}`, {
   118→      method: 'PATCH',
   119→      headers: { 'Content-Type': 'application/json' },
   120→      body: JSON.stringify({ status }),
   121→    })
   122→    const data = await res.json()
   123→    if (!res.ok) {
   124→      toast({ description: data.error ?? t('aiError'), variant: 'destructive' })
   125→      return
   126→    }
   127→    setDetail(data as IdeaDetail)
   128→    refreshIdeas()
   129→  }
   130→
   131→  async function confirmAll(kind: Kind) {
   132→    setBusyKind(kind)
   133→    const list =
   134→      kind === 'processes' ? processes : kind === 'entities' ? entities : kind === 'apis' ? apis : kind === 'screens' ? screens : []
   135→    const artifactKind = kind === 'processes' ? 'process' : kind === 'entities' ? 'entity' : kind === 'apis' ? 'api' : 'screen'
   136→    for (const item of list) {
   137→      if (item.status === 'proposed' || item.status === 'needs_revision') {
   138→        const res = await fetch(`/api/artifacts/${artifactKind}/${item.id}`, {
   139→          method: 'PATCH',
   140→          headers: { 'Content-Type': 'application/json' },
   141→          body: JSON.stringify({ status: 'confirmed' }),
   142→        })
   143→        if (res.ok) setDetail((await res.json()) as IdeaDetail)
   144→      }
   145→    }
   146→    setBusyKind(null)
   147→    refreshIdeas()
   148→  }
   149→
   150→  async function passGate() {
   151→    setBusyGate(true)
   152→    const res = await fetch(`/api/ideas/${idea.id}/gate`, {
   153→      method: 'POST',
   154→      headers: { 'Content-Type': 'application/json' },
   155→      body: JSON.stringify({ gate: 'design', action: 'approve' }),
   156→    })
   157→    const data = await res.json()
   158→    setBusyGate(false)
   159→    if (!res.ok) {
   160→      toast({ description: data.error ?? t('aiError'), variant: 'destructive' })
   161→      return
   162→    }
   163→    setDetail(data as IdeaDetail)
   164→    refreshIdeas()
   165→  }
   166→
   167→  // NO LIMITS: no locked teaser — the stage renders its generate
   168→  // bars and full content at ANY stage, even when empty.
   169→
   170→  return (
   171→    <div className="space-y-4">
   172→      <p className="text-xs text-muted-foreground leading-6">{t('designIntro')}</p>
   173→
   174→      <Tabs value={tab} onValueChange={setTab}>
   175→        <TabsList className="w-full justify-start overflow-x-auto nice-scroll h-auto p-1">
   176→          <TabsTrigger value="processes" className="gap-1.5 text-xs">
   177→            <Workflow className="h-3.5 w-3.5" />
   178→            {t('tabProcesses')} ({processes.length})
   179→          </TabsTrigger>
   180→          <TabsTrigger value="entities" className="gap-1.5 text-xs">
   181→            <Database className="h-3.5 w-3.5" />
   182→            {t('tabData')} ({entities.length})
   183→          </TabsTrigger>
   184→          <TabsTrigger value="apis" className="gap-1.5 text-xs">
   185→            <Server className="h-3.5 w-3.5" />
   186→            {t('tabApis')} ({apis.length})
   187→          </TabsTrigger>
   188→          <TabsTrigger value="screens" className="gap-1.5 text-xs">
   189→            <LayoutTemplate className="h-3.5 w-3.5" />
   190→            {t('tabScreens')} ({screens.length})
   191→          </TabsTrigger>
   192→          <TabsTrigger value="architecture" className="gap-1.5 text-xs">
   193→            <Boxes className="h-3.5 w-3.5" />
   194→            {t('tabArchitecture')}
   195→          </TabsTrigger>
   196→        </TabsList>
   197→
   198→        {/* ---------- Processes ---------- */}
   199→        <TabsContent value="processes" className="mt-3 space-y-3">
   200→          <GenerateBar kind="processes" count={processes.length} unreviewed={processes.filter((p) => p.status === 'proposed' || p.status === 'needs_revision').length} rejected={processes.filter((p) => p.status === 'rejected').length} active={active} busy={busyKind === 'processes'} t={t} onGenerate={() => generate('processes', feedback.processes)} onConfirmAll={() => confirmAll('processes')} />
   201→          <FeedbackPanel
   202→            active={active && processes.length > 0}
   203→            busy={busyKind === 'processes'}
   204→            value={feedback.processes}
   205→            onChange={(v) => setFeedback((f) => ({ ...f, processes: v }))}
   206→            t={t}
   207→            onRun={() => generate('processes', feedback.processes)}
   208→          />
   209→          {processes.map((p) => (
   210→            <div key={p.id} className={`rounded-xl border bg-card p-3 sm:p-4 ${p.status === 'rejected' ? 'opacity-60' : ''}`}>
   211→              <div className="flex items-start gap-2 flex-wrap">
   212→                <CodeChip code={p.code} className="mt-0.5" />
   213→                <div className="flex-1 min-w-[200px]">
   214→                  <div className="text-sm font-medium" dir="ltr">
   215→                    {p.name}
   216→                    {p.nameAr && <span className="text-muted-foreground font-arabic ms-2" dir="rtl">· {p.nameAr}</span>}
   217→                  </div>
   218→                  {p.trigger && <p className="text-xs text-muted-foreground mt-0.5" dir="ltr">⚡ {p.trigger}</p>}
   219→                  <div className="flex flex-wrap gap-1 mt-1.5">
   220→                    {p.actors.map((a) => (
   221→                      <Badge key={a} variant="outline" className="text-[10px] bg-muted/40">{a}</Badge>
   222→                    ))}
   223→                    {p.linkedRequirements.map((c) => <CodeChip key={c} code={c} />)}
   224→                  </div>
   225→                </div>
   226→                <ReviewBadge status={p.status} />
   227→              </div>
   228→              {expanded === p.id && (
   229→                <div className="mt-3 border-t pt-3 space-y-3" dir="ltr">
   230→                  <div className="overflow-x-auto nice-scroll">
   231→                    <table className="w-full text-xs">
   232→                      <thead>
   233→                        <tr className="text-muted-foreground border-b">
   234→                          <th className="py-1.5 text-start font-medium pe-2">#</th>
   235→                          <th className="py-1.5 text-start font-medium pe-2">Actor</th>
   236→                          <th className="py-1.5 text-start font-medium pe-2">Action</th>
   237→                          <th className="py-1.5 text-start font-medium pe-2">Decision / Exception</th>
   238→                        </tr>
   239→                      </thead>
   240→                      <tbody>
   241→                        {p.steps.map((s) => (
   242→                          <tr key={s.n} className="border-b last:border-0 align-top">
   243→                            <td className="py-1.5 pe-2 font-mono text-muted-foreground">{s.n}</td>
   244→                            <td className="py-1.5 pe-2 whitespace-nowrap">{s.actor}</td>
   245→                            <td className="py-1.5 pe-2">{s.action}{s.input ? <span className="text-muted-foreground"> (in: {s.input})</span> : ''}{s.output ? <span className="text-muted-foreground"> (out: {s.output})</span> : ''}</td>
   246→                            <td className="py-1.5 text-muted-foreground">{s.decision ?? s.exception ?? '—'}</td>
   247→                          </tr>
   248→                        ))}
   249→                      </tbody>
   250→                    </table>
   251→                  </div>
   252→                  {p.businessRules.length > 0 && (
   253→                    <div>
   254→                      <div className="text-[11px] font-semibold text-muted-foreground mb-1">{t('businessRules')}</div>
   255→                      <ul className="space-y-1">
   256→                        {p.businessRules.map((b, i) => (
   257→                          <li key={i} className="text-xs leading-6 flex gap-1.5 flex-wrap">
   258→                            <GitBranch className="h-3.5 w-3.5 text-primary shrink-0 mt-1" />
   259→                            <span>
   260→                              {b.rule}
   261→                              {b.linkedRequirements.length > 0 && <span className="ms-1">{b.linkedRequirements.map((c) => <CodeChip key={c} code={c} className="ms-1" />)}</span>}
   262→                            </span>
   263→                          </li>
   264→                        ))}
   265→                      </ul>
   266→                    </div>
   267→                  )}
   268→                  {p.exceptions.length > 0 && (
   269→                    <div>
   270→                      <div className="text-[11px] font-semibold text-muted-foreground mb-1">{t('exceptionsTitle')}</div>
   271→                      <ul className="space-y-0.5">
   272→                        {p.exceptions.map((e, i) => (
   273→                          <li key={i} className="text-xs leading-6 text-muted-foreground">• {e}</li>
   274→                        ))}
   275→                      </ul>
   276→                    </div>
   277→                  )}
   278→                </div>
   279→              )}
   280→              <Expand id={p.id} expanded={expanded} setExpanded={setExpanded} t={t} />
   281→              <ReviewActions kind="process" id={p.id} status={p.status} active={active} t={t} onReview={review} />
   282→            </div>
   283→          ))}
   284→        </TabsContent>
   285→
   286→        {/* ---------- Entities ---------- */}
   287→        <TabsContent value="entities" className="mt-3 space-y-3">
   288→          <GenerateBar kind="entities" count={entities.length} unreviewed={entities.filter((e) => e.status === 'proposed' || e.status === 'needs_revision').length} rejected={entities.filter((e) => e.status === 'rejected').length} active={active} busy={busyKind === 'entities'} t={t} onGenerate={() => generate('entities', feedback.entities)} onConfirmAll={() => confirmAll('entities')} />
   289→          <FeedbackPanel
   290→            active={active && entities.length > 0}
   291→            busy={busyKind === 'entities'}
   292→            value={feedback.entities}
   293→            onChange={(v) => setFeedback((f) => ({ ...f, entities: v }))}
   294→            t={t}
   295→            onRun={() => generate('entities', feedback.entities)}
   296→          />
   297→          {entities.map((e) => (
   298→            <div key={e.id} className={`rounded-xl border bg-card p-3 sm:p-4 ${e.status === 'rejected' ? 'opacity-60' : ''}`}>
   299→              <div className="flex items-start gap-2 flex-wrap">
   300→                <CodeChip code={e.code} className="mt-0.5" />
   301→                <div className="flex-1 min-w-[200px]">
   302→                  <div className="text-sm font-medium" dir="ltr">
   303→                    {e.name}
   304→                    {e.nameAr && <span className="text-muted-foreground font-arabic ms-2" dir="rtl">· {e.nameAr}</span>}
   305→                  </div>
   306→                  {e.description && <p className="text-xs text-muted-foreground mt-0.5 line-clamp-2" dir="ltr">{e.description}</p>}
   307→                  <div className="text-[11px] text-muted-foreground mt-1">{e.fields.length} {t('fieldsTitle').toLowerCase()}</div>
   308→                </div>
   309→                <ReviewBadge status={e.status} />
   310→              </div>
   311→              {expanded === e.id && (
   312→                <div className="mt-3 border-t pt-3 space-y-3" dir="ltr">
   313→                  <div className="overflow-x-auto nice-scroll">
   314→                    <table className="w-full text-xs">
   315→                      <thead>
   316→                        <tr className="text-muted-foreground border-b">
   317→                          <th className="py-1.5 text-start font-medium pe-2">Field</th>
   318→                          <th className="py-1.5 text-start font-medium pe-2">Type</th>
   319→                          <th className="py-1.5 text-start font-medium pe-2">R</th>
   320→                          <th className="py-1.5 text-start font-medium pe-2">U</th>
   321→                          <th className="py-1.5 text-start font-medium">Description</th>
   322→                        </tr>
   323→                      </thead>
   324→                      <tbody>
   325→                        {e.fields.map((f) => (
   326→                          <tr key={f.name} className="border-b last:border-0">
   327→                            <td className="py-1.5 pe-2 font-mono">{f.name}</td>
   328→                            <td className="py-1.5 pe-2 text-muted-foreground">{f.type}</td>
   329→                            <td className="py-1.5 pe-2">{f.required ? '✓' : ''}</td>
   330→                            <td className="py-1.5 pe-2">{f.unique ? '✓' : ''}</td>
   331→                            <td className="py-1.5 text-muted-foreground">{f.description}{f.refEntity ? <span className="text-primary"> → {f.refEntity}</span> : ''}</td>
   332→                          </tr>
   333→                        ))}
   334→                      </tbody>
   335→                    </table>
   336→                  </div>
   337→                  {e.relationships.length > 0 && (
   338→                    <div className="text-xs text-muted-foreground">
   339→                      <span className="font-semibold text-foreground">{t('relationshipsTitle')}: </span>
   340→                      {e.relationships.map((r, i) => (
   341→                        <span key={i}>{i > 0 && ' · '}{r.type} → {r.target} <span className="text-muted-foreground/70">({r.description})</span></span>
   342→                      ))}
   343→                    </div>
   344→                  )}
   345→                  {e.constraints.length > 0 && (
   346→                    <div className="text-xs">
   347→                      <span className="font-semibold">{t('constraintsTitle')}: </span>
   348→                      <span className="text-muted-foreground">{e.constraints.join(' · ')}</span>
   349→                    </div>
   350→                  )}
   351→                </div>
   352→              )}
   353→              <Expand id={e.id} expanded={expanded} setExpanded={setExpanded} t={t} />
   354→              <ReviewActions kind="entity" id={e.id} status={e.status} active={active} t={t} onReview={review} />
   355→            </div>
   356→          ))}
   357→        </TabsContent>
   358→
   359→        {/* ---------- APIs ---------- */}
   360→        <TabsContent value="apis" className="mt-3 space-y-3">
   361→          <GenerateBar kind="apis" count={apis.length} unreviewed={apis.filter((a) => a.status === 'proposed' || a.status === 'needs_revision').length} rejected={apis.filter((a) => a.status === 'rejected').length} active={active} busy={busyKind === 'apis'} t={t} onGenerate={() => generate('apis', feedback.apis)} onConfirmAll={() => confirmAll('apis')} />
   362→          <FeedbackPanel
   363→            active={active && apis.length > 0}
   364→            busy={busyKind === 'apis'}
   365→            value={feedback.apis}
   366→            onChange={(v) => setFeedback((f) => ({ ...f, apis: v }))}
   367→            t={t}
   368→            onRun={() => generate('apis', feedback.apis)}
   369→          />
   370→          {apis.map((a) => (
   371→            <div key={a.id} className={`rounded-xl border bg-card p-3 sm:p-4 ${a.status === 'rejected' ? 'opacity-60' : ''}`}>
   372→              <div className="flex items-start gap-2 flex-wrap">
   373→                <CodeChip code={a.code} className="mt-0.5" />
   374→                <div className="flex-1 min-w-[200px]">
   375→                  <div className="text-sm font-mono font-medium" dir="ltr">
   376→                    <Badge className={`me-1.5 text-[10px] ${a.method === 'GET' ? 'bg-teal-600' : a.method === 'POST' ? 'bg-primary' : a.method === 'DELETE' ? 'bg-red-500' : 'bg-amber-500'}`}>
   377→                      {a.method}
   378→                    </Badge>
   379→                    {a.path}
   380→                  </div>
   381→                  <p className="text-xs text-muted-foreground mt-1 line-clamp-2" dir="ltr">
   382→                    {a.auth && <span className="font-semibold">{a.auth}: </span>}
   383→                    {a.purpose}
   384→                  </p>
   385→                  <div className="flex flex-wrap gap-1 mt-1.5">
   386→                    {a.linkedRequirements.map((c) => <CodeChip key={c} code={c} />)}
   387→                    {a.linkedEntities.map((c) => <CodeChip key={c} code={c} className="bg-teal-50 border-teal-200 text-teal-700" />)}
   388→                  </div>
   389→                </div>
   390→                <ReviewBadge status={a.status} />
   391→              </div>
   392→              {expanded === a.id && (
   393→                <div className="mt-3 border-t pt-3 space-y-3" dir="ltr">
   394→                  {a.request.length > 0 && (
   395→                    <div>
   396→                      <div className="text-[11px] font-semibold text-muted-foreground mb-1">{t('requestFields')}</div>
   397→                      <div className="overflow-x-auto nice-scroll">
   398→                        <table className="w-full text-xs">
   399→                          <thead>
   400→                            <tr className="text-muted-foreground border-b">
   401→                              <th className="py-1.5 text-start font-medium pe-2">Field</th>
   402→                              <th className="py-1.5 text-start font-medium pe-2">Type</th>
   403→                              <th className="py-1.5 text-start font-medium">Validation</th>
   404→                            </tr>
   405→                          </thead>
   406→                          <tbody>
   407→                            {a.request.map((f) => (
   408→                              <tr key={f.field} className="border-b last:border-0">
   409→                                <td className="py-1.5 pe-2 font-mono">{f.field}{f.required ? <span className="text-red-500">*</span> : ''}</td>
   410→                                <td className="py-1.5 pe-2 text-muted-foreground">{f.type}</td>
   411→                                <td className="py-1.5 text-muted-foreground">{f.notes}</td>
   412→                              </tr>
   413→                            ))}
   414→                          </tbody>
   415→                        </table>
   416→                      </div>
   417→                    </div>
   418→                  )}
   419→                  {a.errors.length > 0 && (
   420→                    <div className="text-xs">
   421→                      <span className="font-semibold">{t('errorCodes')}: </span>
   422→                      {a.errors.map((e, i) => (
   423→                        <span key={i}>{i > 0 && ' · '}<code className="bg-muted/60 rounded px-1 py-0.5 font-mono text-[10px]">{e.code}</code> {e.when}</span>
   424→                      ))}
   425→                    </div>
   426→                  )}
   427→                </div>
   428→              )}
   429→              <Expand id={a.id} expanded={expanded} setExpanded={setExpanded} t={t} />
   430→              <ReviewActions kind="api" id={a.id} status={a.status} active={active} t={t} onReview={review} />
   431→            </div>
   432→          ))}

   728→function GenerateBar({
   729→  kind,
   730→  count,
   731→  unreviewed,
   732→  rejected,
   733→  active,
   734→  busy,
   735→  t,
   736→  onGenerate,
   737→  onConfirmAll,
   738→}: {
   739→  kind: Kind
   740→  count: number
   741→  unreviewed: number
   742→  rejected: number
   743→  active: boolean
   744→  busy: boolean
   745→  t: (key: Parameters<ReturnType<typeof useApp>['t']>[0]) => string
   746→  onGenerate: () => void
   747→  onConfirmAll: (() => void) | null
   748→}) {
   749→  return (
   750→    <div className="flex flex-wrap items-center gap-2">
   751→      {busy ? (
   752→        <WorkingPanel steps={[t(GENERATE_KEYS[kind].steps[0]), t(GENERATE_KEYS[kind].steps[1])]} className="flex-1" />

   777→function ReviewActions({
   778→  kind,
   779→  id,
   780→  status,
   781→  active,
   782→  t,
   783→  onReview,
   784→}: {
   785→  kind: string
   786→  id: string
   787→  status: string
   788→  active: boolean
   789→  t: (key: Parameters<ReturnType<typeof useApp>['t']>[0]) => string
   790→  onReview: (kind: string, id: string, status: string) => void
   791→}) {
   792→  // needs_revision items stay decidable even after the stage advanced —
   793→  // change-impact control must never become a dead end.
   794→  const decidable = active || status === 'needs_revision'
   795→  if (!decidable || status === 'rejected') return null
   796→  return (
   797→    <div className="mt-2 flex gap-1.5 border-t pt-2">
   798→      {status !== 'confirmed' && (
   799→        <Button size="sm" variant="outline" className="gap-1 border-emerald-300 text-emerald-700 hover:bg-emerald-50" onClick={() => onReview(kind, id, 'confirmed')}>
   800→          <Check className="h-3.5 w-3.5" />
   801→          {t('confirmArtifact')}
   802→        </Button>
   803→      )}
   804→      {status !== 'rejected' && (
   805→        <Button size="sm" variant="outline" className="gap-1 border-red-200 text-red-600 hover:bg-red-50" onClick={() => onReview(kind, id, 'rejected')}>
   806→          <X className="h-3.5 w-3.5" />
   807→          {t('rejectArtifact')}
   808→        </Button>
   809→      )}
   810→    </div>
   811→  )
   812→}
